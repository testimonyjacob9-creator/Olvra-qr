match /love/{id} {
  allow get: if true;
  allow list, update, delete: if false;
  allow create: if id.size() >= 12
    && request.resource.data.keys().hasOnly(['to','from','msg','anim','theme','at'])
    && request.resource.data.to is string && request.resource.data.to.size() > 0 && request.resource.data.to.size() <= 40
    && request.resource.data.from is string && request.resource.data.from.size() > 0 && request.resource.data.from.size() <= 40
    && request.resource.data.msg is string && request.resource.data.msg.size() > 0 && request.resource.data.msg.size() <= 600
    && request.resource.data.anim in ['hearts','confetti','typing']
    && request.resource.data.theme in [0,1,2]
    && request.resource.data.at == request.time;
}
match /loveReports/{rid} {
  allow create: if request.resource.data.keys().hasOnly(['id','at'])
    && request.resource.data.id is string && request.resource.data.id.size() <= 20
    && request.resource.data.at == request.time;
  allow read, update, delete: if false;
}
